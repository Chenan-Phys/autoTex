\begin{frame}[allowframebreaks]
\frametitle{Summary}
    \begin{center}
        \begin{minipage}{0.8\textwidth}
        \begin{block}{Disordered kagome lattice}
            \begin{itemize}
                \item Onsite disorder leads to a loop flatband states
                \item Interacting loop flatband states can be described by SYK model
            \end{itemize}
        \end{block}
        \end{minipage}
    
        \vspace{0.5cm}
    
        \begin{minipage}{0.8\textwidth}
        \begin{block}{Disordered MATBG}
            \begin{itemize}
                \item Interplay between MATBG flatband and disorder leads to flatband states circulating around the boundary
                \item Interacting of such states gives rise to rich phases, such as chaotic stange metal, superconductivity, correlated insulator
            \end{itemize}
        \end{block}
        \end{minipage}
    
        \vspace{1cm}
    
        \begin{minipage}{0.8\textwidth}
        \begin{block}{Chiral Spin Liquid}
            \begin{itemize}
                \item Next-nearest-neighbor hoppings can give rise to moat band
                \item Spins/Bosons in moat band form topological non-trivial ground states
                \item Such states can be described by Chern-Simon field theory
            \end{itemize}
        \end{block}
        \end{minipage}
    
        \vspace{0.5cm}
    
        \begin{minipage}{0.8\textwidth}
        \begin{block}{Integrable Chiral Spin Chain}
            \begin{itemize}
                \item Chiral spin chain shows a critical point between gapless states
                \item The finite size correction of such critical point shows a universal non-monotonic behavior
            \end{itemize}
        \end{block}
        \end{minipage}
    \end{center}
\end{frame}
